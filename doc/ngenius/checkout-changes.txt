    /** The external folder, checked at startup; null when not configured (then only full locations work). */
    private static Path certificateDir(String dir) {
        if (dir == null || dir.isBlank()) {
            return null;
        }
        Path path = Path.of(dir).toAbsolutePath().normalize();
        if (!Files.isDirectory(path)) {
            throw new IllegalStateException("wallet.apple-pay.certificate-dir (APPLE_PAY_CERT_DIR) is not a folder: "
                    + path);
        }
        if (!Files.isReadable(path)) {
            throw new IllegalStateException("wallet.apple-pay.certificate-dir is not readable by the application user: "
                    + path);
        }
        log.info("Apple Pay certificate folder: {}", path);
        return path;
    }

    /**
     * Where a file is:
     *   "classpath:apple/x.cer", "file:/etc/x.p12"  -> as written
     *   "/opt/certs/x.p12" (absolute path)          -> that file
     *   "apple-pay.p12" (plain name)                -> certificate-dir/apple-pay.p12
     */
    private static Resource resource(String location, Path certificateDir, ResourceLoader resourceLoader) {
        if (location == null || location.isBlank()) {
            throw new IllegalStateException("Apple Pay file location is empty (keystore / root-ca-certificate)");
        }
        if (location.startsWith("classpath:") || location.startsWith("file:")) {
            return resourceLoader.getResource(location);
        }
        Path path = Path.of(location);
        if (!path.isAbsolute()) {
            if (certificateDir == null) {
                throw new IllegalStateException("'" + location + "' is a plain file name but "
                        + "wallet.apple-pay.certificate-dir (APPLE_PAY_CERT_DIR) is not set");
            }
            path = certificateDir.resolve(location).normalize();
        }
        return new FileSystemResource(path);
    }

    private static String describe(Resource resource) {
        try {
            return resource.getFile().getAbsolutePath();
        } catch (IOException ex) {
            return resource.getDescription();   // classpath resource inside the jar
        }
    }

1. keys/WalletKeys.java: add two readers (plain JDK)

java
    /** Apple's "merchant identifier" certificate extension: SHA-256 of the merchant id. */
    public static final String APPLE_MERCHANT_ID_OID = "1.2.840.113635.100.6.32";

    /**
     * SHA-256 of the merchant id, read from the Apple Pay certificate (32 bytes), or null if the certificate
     * has no such extension (then it is not an Apple Pay certificate).
     * Apple stores it as a hex string (UTF8String / IA5String); a raw 32-byte OCTET STRING is accepted too.
     */
    public static byte[] appleMerchantIdentifierHash(X509Certificate certificate) {
        byte[] extension = certificate.getExtensionValue(APPLE_MERCHANT_ID_OID);
        if (extension == null) {
            return null;
        }
        byte[] inner = derValue(extension);          // the extension value is wrapped in an OCTET STRING
        int tag = inner[0] & 0xff;
        byte[] value = derValue(inner);
        if (tag == 0x04 && value.length == 32) {
            return value;
        }
        byte[] hash = HexFormat.of().parseHex(new String(value, StandardCharsets.US_ASCII).trim());
        if (hash.length != 32) {
            throw new IllegalStateException("Apple merchant identifier extension is not a SHA-256 hash");
        }
        return hash;
    }

    /** The plain merchant id (subject UID, e.g. merchant.com.company.shop), for logs; null if absent. */
    public static String appleMerchantIdentifier(X509Certificate certificate) {
        String dn = certificate.getSubjectX500Principal().getName(X500Principal.RFC2253);
        for (String part : dn.split("(?<!\\\\),")) {
            if (part.startsWith("UID=")) {
                return part.substring(4);
            }
        }
        return null;
    }

    /** Value bytes of one DER element (tag, short or long-form length, value). */
    private static byte[] derValue(byte[] der) {
        int i = 1;
        int length = der[i++] & 0xff;
        if (length > 0x7f) {
            int bytes = length & 0x7f;
            length = 0;
            for (int k = 0; k < bytes; k++) {
                length = (length << 8) | (der[i++] & 0xff);
            }
        }
        return Arrays.copyOfRange(der, i, i + length);
    }

     * The merchant id hash comes from the certificate. wallet.apple-pay.merchant-identifier is optional:
     * if set, it must match the certificate (catches a .p12 from the wrong merchant at startup).

    private static byte[] merchantIdentifierHash(X509Certificate certificate, String configuredMerchantId) {
        byte[] fromCertificate = WalletKeys.appleMerchantIdentifierHash(certificate);
        if (fromCertificate == null) {
            throw new IllegalStateException("Certificate " + certificate.getSubjectX500Principal().getName()

                    + "' does not match the certificate's merchant ("
                    + WalletKeys.appleMerchantIdentifier(certificate) + ")");

    }

java
    /** Masks a JSON document by path. Invalid JSON is not returned at all (it could hide anything). */
    public String maskJson(String json) {
        if (json == null || json.isBlank()) {
            return json;
        }
        try {
            JsonNode root = JSON.readTree(json);
            maskChildren(root, new ArrayList<>());
            return JSON.writeValueAsString(root);
        } catch (JacksonException ex) {
            return "[invalid JSON, " + json.length() + " chars, not logged]";
        }
    }

    /**
     * Walks the tree with the current path (lower-case field names; arrays add no step).
     * A field whose path matches a rule is replaced with its masked text; its children are not visited.
     */
    private void maskChildren(JsonNode node, List<String> path) {
        if (node instanceof ObjectNode object) {
            for (Map.Entry<String, JsonNode> field : new ArrayList<>(object.properties())) {
                path.add(field.getKey().toLowerCase(Locale.ROOT));
                MaskStrategy strategy = strategyFor(path);
                JsonNode value = field.getValue();
                if (strategy == null) {
                    maskChildren(value, path);
                } else if (!value.isNull()) {                  // null stays null: "was it sent?" stays visible
                    object.put(field.getKey(), maskedText(value, strategy));
                }
                path.removeLast();
            }
        } else if (node instanceof ArrayNode array) {
            for (JsonNode element : array) {
                maskChildren(element, path);
            }
        }
    }

    /** Objects and arrays -> "***"; strings and numbers -> the strategy applied to their text. */
    private static String maskedText(JsonNode value, MaskStrategy strategy) {
        if (value.isContainer()) {
            return MASK;
        }
        return strategy.apply(value.isString() ? value.stringValue() : value.asString());
    }

How it works:

Read: readTree turns the body into a tree.
Walk: maskChildren goes through it, keeping the current path. When a field’s path matches a rule, object.put(...) replaces its value with the masked string.
Write: writeValueAsString turns the tree back into a string.
Arrays: they don’t add a step to the path, so items.card.number still matches every element.
Bad JSON: invalid JSON, nesting deeper than 64 levels, or extra content after the document makes readTree throw, so the body is not logged.

If isString(), stringValue() or asString() don’t compile, your Jackson version uses the older names isTextual(), textValue() and asText().
