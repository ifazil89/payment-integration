A Checkout.com payment response includes source.id (src_…), a stored reference to the card. You save it and pay with it later using "source": {"type": "id", "id": "src_…"}. That covers both a shopper paying with a saved card and a merchant-initiated charge (recurring or unscheduled) that links back to the first payment. I haven’t compiled or run these changes.

package com.company.checkout.payment.dto;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;

/**
 * Pay with a card Checkout.com stored on an earlier payment (PaymentResponse.sourceId).
 *
 * Shopper present (merchantInitiated false/absent): cvv optional, 3DS like a plain card.
 * Merchant-initiated (merchantInitiated true): no cvv, no 3DS; paymentType and previousPaymentId required.
 *
 * @param previousPaymentId the Checkout.com id (pay_...) of the first payment with this card (links the MIT
 *                          to the shopper-authenticated one, as the card schemes require)
 */
public record StoredCardDetails(
        @NotBlank @Pattern(regexp = "^src_[a-z0-9]{26}$", message = "must be a Checkout.com source id (src_...)")
        String sourceId,
        @Pattern(regexp = "\\d{3,4}", message = "must be 3 or 4 digits") String cvv,
        Boolean merchantInitiated,
        MitPaymentType paymentType,
        @Pattern(regexp = "^pay_[a-z0-9]{26}$", message = "must be a Checkout.com payment id (pay_...)")
        String previousPaymentId) {

    /** Checkout.com payment_type for merchant-initiated payments. */
    public enum MitPaymentType {
        RECURRING("Recurring"), UNSCHEDULED("Unscheduled"), INSTALLMENT("Installment");

        private final String cko;

        MitPaymentType(String cko) {
            this.cko = cko;
        }

        public String cko() {
            return cko;
        }
    }

    public boolean isMerchantInitiated() {
        return Boolean.TRUE.equals(merchantInitiated);
    }

    @Override
    public String toString() {
        return "StoredCardDetails[sourceId=****" + (sourceId == null ? null : sourceId.substring(Math.max(0, sourceId.length() - 4)))
                + ", cvv=" + (cvv == null ? null : "***") + ", merchantInitiated=" + merchantInitiated
                + ", paymentType=" + paymentType + ", previousPaymentId=" + previousPaymentId + "]";
    }
}

Check the exact src_ and pay_ formats against your sandbox responses. If they differ, relax the two patterns to ^src_\w+$ and ^pay_\w+$.
